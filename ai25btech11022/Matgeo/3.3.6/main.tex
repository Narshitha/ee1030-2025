\textbf{Question}:\\
 Construct a triangle ABC with side $\vec{BC}=6cm,\angle{B}=45\degree,\angle{A}=105\degree$ \\
 \textbf{Solution}:\\
 Let position vector of $\vec{B}$ be $\myvec{0\\0}$ and a,b,c be the sides opposite to the vertices A,B and C\\
 Given a=6cm;
 \begin{align}
     \vec{C}=\myvec{6\\0}\\
     \vec{A}=\myvec{c\cos{\angle{B}}\\c\sin{\angle{B}}}\\
     \vec{A}=\myvec{\frac{c}{\sqrt{2}}\\\frac{c}{\sqrt{2}}}\\
     \angle{A}+\angle{B}+\angle{C}=180\degree\\
     \therefore \angle{C}=30\degree
 \end{align}
 in $\Delta ABC$
 \begin{align}
     b\cos{\angle{C}} +c\cos{\angle{B}} =a\\
     b\sin{\angle{C}}-c\sin{\angle{B}} =0\\
     \myvec{\cos{\angle{C}} && \cos{\angle{B}}\\ \sin{\angle{C}} && -\sin{\angle{B}}}\myvec{b\\c}=\myvec{6\\0}
 \end{align}
 using augmented matrix 
 \begin{align}
     \myvec{\cos{\angle{C}} && \cos{\angle{B}} && \vrule && 6\\ \sin{\angle{C}} && -\sin{\angle{B}} && \vrule && 0}\\
     \myvec{\sqrt{3}/2 && 1/\sqrt{2}&& \vrule && 6 \\1/2 && -1/\sqrt{2} &&  \vrule && 0}
 \end{align}
 $R_2 \longrightarrow \sqrt{3}R_2 -R_1$
 \begin{align}
     \myvec{\sqrt{3}/2 && 1/\sqrt{2}&& \vrule && 6 \\0 && -\sqrt{3}-1/\sqrt{2} &&  \vrule && -6}\\
     \implies c= \frac{6\sqrt{2}}{\sqrt{3}+1}\\
   \therefore  \vec{A}=\myvec{6\brak{\sqrt{3}-1}\\6\brak{\sqrt{3}-1}}
 \end{align}
