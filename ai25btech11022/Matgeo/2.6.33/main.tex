\textbf{Question}:\\
Find area of the triangle with vertices at point given in each of the following:\\
    a) \brak{1,0},\brak{6,0},\brak{4,3}\\
    b) \brak{2,7},\brak{1,1},\brak{10,8}\\
    c) \brak{-2,-3},\brak{3,2},\brak{-1,8}\\
\textbf{Solution}:\\
\begin{tabular}{|c|c|c|c|} \hline
   & $\vec{A-B}$ & $\vec{A-C}$ & $\frac{1}{2}\|\vec{A-B} \times \vec{A-C}\|$\\ \hline
    a) & $\myvec{-5\\0}$ & $\myvec{-3\\-3}$ & $7.5$\\ \hline
    b) & $\myvec{1\\6}$ & $\myvec{-8\\-1}$ & $23.5$\\ \hline
    c) & $\myvec{-5\\-5}$ & $\myvec{-1\\-11}$ & $25$\\ \hline 
\end{tabular}
